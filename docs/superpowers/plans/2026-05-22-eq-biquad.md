**Goal:** Implement `FilterSpec::Eq` in both Python and Rust using per-bin complex biquad frequency response, enabling the `known_eq_preset` fixture parity test to pass.

**Architecture:** Both Python and Rust switch from time-domain IIR filtering to a spectral approach: compute biquad coefficients analytically, evaluate H(e^jω) at each RFFT bin, and apply as multiplicative complex gain. The STFT path in Rust replaces the `chain_scalar` pattern with per-bin complex filter application for EQ chains; the wavelet/non-STFT path wraps the contribution signal with rfft→apply→irfft.

```python
def _eq_frequency_response(spec: FilterSpec, sample_rate: int, n: int) -> np.ndarray:
    """Return complex frequency response array of shape (n//2 + 1,) for a biquad EQ spec."""
    mode = str(spec.parameters.get("mode", "peaking"))
    frequency = float(spec.parameters["frequency_hz"])
    q = float(spec.parameters.get("q", 0.707))
    gain_db = float(spec.parameters.get("gain_db", 0.0))

    num_bins = n // 2 + 1
    # Bin frequencies: omega_k = 2*pi*k / n  (k = 0 .. num_bins-1)
    k = np.arange(num_bins, dtype=np.float64)
    omega = 2.0 * np.pi * k / n
    z_inv = np.exp(-1j * omega)          # e^{-j*omega} per bin
    z_inv2 = np.exp(-2j * omega)         # e^{-2j*omega} per bin

    if mode == "highpass":
        # 2nd-order Butterworth highpass via bilinear transform
        fc = float(frequency)
        fs = float(sample_rate)
        kw = np.tan(np.pi * fc / fs)     # pre-warped angular frequency
        denom = 1.0 + np.sqrt(2.0) * kw + kw ** 2
        b0 =  1.0 / denom
        b1 = -2.0 * b0
        b2 =  b0
        a1 =  2.0 * (kw ** 2 - 1.0) * b0
        a2 =  (1.0 - np.sqrt(2.0) * kw + kw ** 2) * b0
        h = (b0 + b1 * z_inv + b2 * z_inv2) / (1.0 + a1 * z_inv + a2 * z_inv2)

    elif mode == "lowpass":
        # 2nd-order Butterworth lowpass via bilinear transform
        fc = float(frequency)
        fs = float(sample_rate)
        kw = np.tan(np.pi * fc / fs)
        denom = 1.0 + np.sqrt(2.0) * kw + kw ** 2
        b0 = kw ** 2 / denom
        b1 = 2.0 * b0
        b2 = b0
        a1 = 2.0 * (kw ** 2 - 1.0) / denom
        a2 = (1.0 - np.sqrt(2.0) * kw + kw ** 2) / denom
        h = (b0 + b1 * z_inv + b2 * z_inv2) / (1.0 + a1 * z_inv + a2 * z_inv2)

    else:
        # Peaking: H_total(omega) = 1 + H_peak(omega) * (A - 1)
        # iirpeak coefficients: w0 = 2*pi*f0/fs, bw = w0/Q
        w0 = 2.0 * np.pi * frequency / sample_rate
        bw = w0 / q
        t_bw2 = np.tan(bw / 2.0)
        # Unnormalized: b = [t_bw2, 0, -t_bw2], a = [1+t_bw2, -2*cos(w0), 1-t_bw2]
        a0 = 1.0 + t_bw2
        b0_peak =  t_bw2 / a0
        b1_peak =  0.0
        b2_peak = -t_bw2 / a0
        a1_peak = -2.0 * np.cos(w0) / a0
        a2_peak = (1.0 - t_bw2) / a0
        h_peak = (b0_peak + b1_peak * z_inv + b2_peak * z_inv2) / (1.0 + a1_peak * z_inv + a2_peak * z_inv2)
        a_lin = 10.0 ** (gain_db / 20.0)
        h = 1.0 + h_peak * (a_lin - 1.0)

    return h


def _apply_eq(audio: np.ndarray, spec: FilterSpec, sample_rate: int) -> np.ndarray:
    n = len(audio)
    h = _eq_frequency_response(spec, sample_rate, n)
    spectrum = np.fft.rfft(audio)
    return np.fft.irfft(spectrum * h, n=n)
```

```rust
/// Compute per-bin complex frequency response H(e^{jω}) for a biquad EQ filter.
///
/// `bins_count` = frame_size / 2 + 1 (number of one-sided RFFT bins).
/// Bin k corresponds to ω_k = 2π·k / (2·(bins_count − 1)).
///
/// Modes:
/// - "highpass": 2nd-order Butterworth highpass via bilinear transform
/// - "lowpass":  2nd-order Butterworth lowpass via bilinear transform
/// - "peaking" (default): iirpeak with H_total = 1 + H_peak·(A − 1)
pub fn eq_frequency_response(
    mode: &str,
    frequency_hz: f64,
    q: f64,
    gain_db: f64,
    sample_rate: u32,
    bins_count: usize,
) -> Vec<Complex<f64>> {
    use std::f64::consts::PI;

    let fs = sample_rate as f64;
    // n = full FFT size = 2 * (bins_count - 1)
    let n = 2 * (bins_count.saturating_sub(1)).max(1);

    let mut h = Vec::with_capacity(bins_count);

    match mode {
        "highpass" => {
            let kw = (PI * frequency_hz / fs).tan();
            let denom = 1.0 + 2.0_f64.sqrt() * kw + kw * kw;
            let b0 = 1.0 / denom;
            let b1 = -2.0 * b0;
            let b2 = b0;
            let a1 = 2.0 * (kw * kw - 1.0) * b0;
            let a2 = (1.0 - 2.0_f64.sqrt() * kw + kw * kw) * b0;

            for k in 0..bins_count {
                let omega = 2.0 * PI * k as f64 / n as f64;
                let z_inv = Complex::from_polar(1.0, -omega);
                let z_inv2 = Complex::from_polar(1.0, -2.0 * omega);
                let num = b0 + b1 * z_inv + b2 * z_inv2;
                let den = Complex::new(1.0, 0.0) + a1 * z_inv + a2 * z_inv2;
                h.push(num / den);
            }
        }
        "lowpass" => {
            let kw = (PI * frequency_hz / fs).tan();
            let denom = 1.0 + 2.0_f64.sqrt() * kw + kw * kw;
            let b0 = kw * kw / denom;
            let b1 = 2.0 * b0;
            let b2 = b0;
            let a1 = 2.0 * (kw * kw - 1.0) / denom;
            let a2 = (1.0 - 2.0_f64.sqrt() * kw + kw * kw) / denom;

            for k in 0..bins_count {
                let omega = 2.0 * PI * k as f64 / n as f64;
                let z_inv = Complex::from_polar(1.0, -omega);
                let z_inv2 = Complex::from_polar(1.0, -2.0 * omega);
                let num = b0 + b1 * z_inv + b2 * z_inv2;
                let den = Complex::new(1.0, 0.0) + a1 * z_inv + a2 * z_inv2;
                h.push(num / den);
            }
        }
        _ => {
            // peaking (default)
            let w0 = 2.0 * PI * frequency_hz / fs;
            let bw = w0 / q;
            let t_bw2 = (bw / 2.0).tan();
            let a0 = 1.0 + t_bw2;
            let b0_peak = t_bw2 / a0;
            let b2_peak = -t_bw2 / a0;
            let a1_peak = -2.0 * w0.cos() / a0;
            let a2_peak = (1.0 - t_bw2) / a0;
            let a_lin = 10.0_f64.powf(gain_db / 20.0);

            for k in 0..bins_count {
                let omega = 2.0 * PI * k as f64 / n as f64;
                let z_inv = Complex::from_polar(1.0, -omega);
                let z_inv2 = Complex::from_polar(1.0, -2.0 * omega);
                // b1_peak = 0, so b0_peak + 0*z_inv + b2_peak*z_inv2
                let num_peak = b0_peak + b2_peak * z_inv2;
                let den_peak = Complex::new(1.0, 0.0) + a1_peak * z_inv + a2_peak * z_inv2;
                let h_peak = num_peak / den_peak;
                h.push(Complex::new(1.0, 0.0) + h_peak * (a_lin - 1.0));
            }
        }
    }

    h
}
```

```rust
/// Apply a filter chain to a mutable slice of complex spectral bins.
/// `sample_rate` and `soloed` are needed for EQ and solo handling respectively.
/// Returns false if the contribution should be zeroed (muted or not soloed).
pub fn apply_chain_to_bins(
    bins: &mut Vec<Complex<f64>>,
    chain: &[FilterSpec],
    sample_rate: u32,
    soloed: &[String],
    name: &str,
) -> bool {
    // If any solo exists and this contribution is not soloed, zero it
    if !soloed.is_empty() && !soloed.iter().any(|s| s == name) {
        return false;
    }

    for spec in chain {
        match spec {
            FilterSpec::Unity | FilterSpec::Solo => {}
            FilterSpec::Gain { db } => {
                let scale = 10.0_f64.powf(db / 20.0);
                for bin in bins.iter_mut() {
                    *bin *= scale;
                }
            }
            FilterSpec::Mute => {
                return false;
            }
```

This function rfft → apply H → irfft for a real signal. Add after `apply_chain_to_bins` in `filters.rs`:

```rust
/// Apply a biquad EQ filter to a real-valued signal using rfft → H(ω) → irfft.
/// Used for the wavelet path where contributions are in the time domain.
pub fn apply_eq_to_real_signal(
    signal: &[f64],
    mode: &str,
    frequency_hz: f64,
    q: f64,
    gain_db: f64,
    sample_rate: u32,
) -> Vec<f64> {
    use rustfft::FftPlanner;

    let n = signal.len();
    if n == 0 {
        return Vec::new();
    }
    let num_bins = n / 2 + 1;

    // Forward FFT (real → complex)
    let mut planner = FftPlanner::new();
    let fft = planner.plan_fft_forward(n);
    let mut buf: Vec<Complex<f64>> = signal.iter().map(|&x| Complex::new(x, 0.0)).collect();
    fft.process(&mut buf);

    // One-sided spectrum
    let mut spectrum: Vec<Complex<f64>> = buf[..num_bins].to_vec();

    // Compute and apply H
    let h = eq_frequency_response(mode, frequency_hz, q, gain_db, sample_rate, num_bins);
    for (bin, hi) in spectrum.iter_mut().zip(h.iter()) {
        *bin *= hi;
    }

    // Reconstruct full two-sided spectrum for inverse FFT
    let mut ifft_buf: Vec<Complex<f64>> = vec![Complex::new(0.0, 0.0); n];
    ifft_buf[..num_bins].copy_from_slice(&spectrum);
    for k in 1..num_bins - 1 {
        ifft_buf[n - k] = spectrum[k].conj();
    }

    let ifft = planner.plan_fft_inverse(n);
    ifft.process(&mut ifft_buf);

    ifft_buf.iter().map(|c| c.re / n as f64).collect()
}
```

```rust
/// Apply a filter chain to a real-valued signal vector.
/// Used by the wavelet path. Returns the filtered signal, or a zero vector if muted/not-soloed.
pub fn apply_chain_to_real_signal(
    signal: &[f64],
    chain: &[FilterSpec],
    sample_rate: u32,
    soloed: &[String],
    name: &str,
) -> Vec<f64> {
    if !soloed.is_empty() && !soloed.iter().any(|s| s == name) {
        return vec![0.0; signal.len()];
    }

    let mut result = signal.to_vec();
    for spec in chain {
        match spec {
            FilterSpec::Unity | FilterSpec::Solo => {}
            FilterSpec::Gain { db } => {
                let scale = 10.0_f64.powf(db / 20.0);
                for s in result.iter_mut() {
                    *s *= scale;
                }
            }
            FilterSpec::Mute => {
                return vec![0.0; signal.len()];
            }
            FilterSpec::Polarity => {
                for s in result.iter_mut() {
                    *s = -*s;
                }
            }
            FilterSpec::Eq { mode, frequency_hz, q, gain_db } => {
                let gdb = gain_db.unwrap_or(0.0);
                result = apply_eq_to_real_signal(&result, mode, *frequency_hz, *q, gdb, sample_rate);
            }
        }
    }
    result
}
```
